/* part:page.script.adopt */
/* The Adopt view (docs/adopt.md). It registers itself with the assistant's app.js through window.ADP_VIEWS.
   Files beside the page, written by `adp.py extract SUBJECT --into DIR`: subject/candidates.json (the analysis) and the subject's own files under subject/.
   The server keeps what the person decided, so a refresh forgets nothing:
     ns/shared/adopt/state   {items: [{id, state, notes, at}]}      ns/shared/adopt/view   {selected, at}: the candidate shown last
     ns/shared/packs/<id>   the adp-pack/1 object of a ready candidate      ns/shared/previews/<id>   {html}: a preview page Claude built for a candidate */
(function(){
'use strict';
const SRC = 'subject/index.html', STATE = 'adopt/state', VIEW = 'adopt/view';
const hhmm = () => { try { return new Date().toLocaleTimeString([], {hour: '2-digit', minute: '2-digit'}); } catch (e) { return ''; } };
const why = e => String((e && (e.code || e.message)) || e || 'unknown error').slice(0, 80);

window.ADP_VIEWS = Object.assign(window.ADP_VIEWS || {}, {
  /* part:adopt */
  adopt: {
    label: 'Adopt',
    parts: ['adopt', 'adopt.intro', 'adopt.board', 'adopt.inspect'],
    titles: {adopt: 'Adopt', 'adopt.intro': 'Adopt intro', 'adopt.board': 'Candidates', 'adopt.inspect': 'Live preview'},
    html: () => `<section class="view adopt-view" data-part="adopt" data-comment-target aria-label="Adopt">
      <!-- part:adopt.intro --><div class="card" data-part="adopt.intro" data-comment-target><h2>Adopt</h2><p class="muted">Analyze, select, work with Claude, preview, then mark each candidate ready or discard it. Pick an element in the preview or a node in the map; a ready candidate downloads as a <code>.adp.json</code> pack. Your decisions are kept on the server.</p><p class="muted ad-note" role="status"></p></div><!-- /part:adopt.intro -->
      <div class="ad-split">
        <!-- part:adopt.board --><div class="card ad-board" data-part="adopt.board" data-comment-target aria-label="Candidates"></div><!-- /part:adopt.board -->
        <!-- part:adopt.inspect --><div class="card ad-inspect" data-part="adopt.inspect" data-comment-target aria-label="Live preview of the artifact"></div><!-- /part:adopt.inspect -->
      </div>
    </section>`,
    mount(ctx){
      const Kit = window.Kit || {}, $ = s => document.querySelector(s), note = $('.ad-note'), boardEl = $('.ad-board'), inspEl = $('.ad-inspect');
      if (!Kit.adopt || !Kit.adoptBoard || !Kit.adoptInspector) { boardEl.innerHTML = '<p class="muted">The adopt kits are not installed.</p>'; return {destroy(){}}; }
      let dead = false, offs = [], pvOff = null, board = null, insp = null, base = [], overlay = {}, packs = {}, sel = null, can = true, restored = false, chain = Promise.resolve();
      /* The runtime connects to the database after the page starts: ADP.db is null until then. Reading it once at mount, as the first releases did, left every database feature
         (decisions, the selection, stored previews) silently off whenever this tab was already open on a refresh. So everything that keeps state waits for the connection. */
      let db = null, ref = null, viewRef = null, diagRef = null;
      const packRef = id => ADP.ns('shared').doc('packs/' + id), previewRef = id => ADP.ns('shared').doc('previews/' + id);      // a doc path has an even number of parts: ns/shared/packs/<id>
      const connect = () => { db = window.ADP && ADP.db; ref = db ? ADP.ns('shared').doc(STATE) : null; viewRef = db ? ADP.ns('shared').doc(VIEW) : null; diagRef = db ? ADP.ns('shared').doc('adopt/diag') : null; };
      const connected = Promise.resolve(window.ADP && ADP.config && ADP.config.ready).catch(() => null).then(() => new Promise(r => { let n = 0; const t = () => { if ((window.ADP && ADP.db) || n++ > 80) { connect(); r(); } else setTimeout(t, 100); }; t(); }));
      note.textContent = 'Connecting to the server…';
      const status = (t, tone) => { if (board) board.setStatus(t, tone); };
      /* What each preview did when it loaded here (ok, blank, error, unreadable), kept in ns/shared/adopt/diag so a session can read what a person sees without asking. */
      const diag = {};
      const reportPreview = s => { diag[s.id] = Object.assign({}, s, {at: new Date().toISOString(), page: String(location.pathname).slice(-80)}); if (diagRef) write(diagRef, {items: Object.keys(diag).map(k => Object.assign({}, diag[k]))}).catch(() => {}); };
      const merged = () => Kit.adopt.merge(base, overlay);
      const push = () => { const l = merged(); if (board) board.setCandidates(l); if (insp) insp.setCandidates(l); };
      /* One write at a time, the whole doc each time. Ids hold dots (c.picker), so the decisions are a list, not a map keyed by id: a db update would read a dotted key as a path. */
      const write = (ref0, data) => { chain = chain.catch(() => {}).then(() => ref0.set(data)); return chain; };
      const items = () => Object.keys(overlay).map(id => Object.assign({id}, overlay[id]));
      const fromItems = d => { const m = {}; ((d && d.items) || []).forEach(x => { if (x && x.id) { const {id, ...rest} = x; m[id] = rest; } }); return m; };
      const save = (id, patch) => {
        overlay = Object.assign({}, overlay, {[id]: Object.assign({}, overlay[id], patch, {at: new Date().toISOString()})}); push();
        if (!ref) { status('Not saved: there is no database here, so this lasts until the page closes.', 'warn'); return Promise.resolve(); }
        status('Saving…', '');
        return write(ref, {items: items()}).then(() => status('Saved to the server at ' + hhmm() + '.', 'ok'), e => status('Not saved: ' + why(e) + '. Reload and try again.', 'danger'));
      };
      const remember = id => { if (viewRef) write(viewRef, {selected: id, at: new Date().toISOString()}).catch(() => {}); };
      /* Each step on its own: a failure in one (the outline, the picture) must never stop the stored preview, the pack or the selection from loading. */
      const choose = (id, fromUser) => {
        sel = id;
        const step = (name, f) => { try { f(); } catch (e) { status('Could not ' + name + ': ' + why(e), 'warn'); } };
        step('load the preview', () => watchPreview(id));
        step('show the candidate', () => { if (board) board.select(id); });
        step('open it in the preview', () => { if (insp) insp.select(id); });
        step('read the pack', () => loadPack(id));
        if (fromUser) step('remember the selection', () => remember(id));
      };
      const loadPack = id => { if (!ref || packs[id] !== undefined) return; packs[id] = null; try { packRef(id).get().then(s => { if (!dead && s && s.exists) { packs[id] = s.data(); if (board) board.setPacks(packs); } }).catch(() => {}); } catch (e) { status('Could not read the pack: ' + why(e), 'warn'); } };
      /* The selected candidate's preview page, if Claude has written one, follows the db: it appears here the moment it is saved. */
      const watchPreview = id => {
        if (pvOff) { try { pvOff(); } catch (e) {} pvOff = null; }
        if (!db) return;
        try { pvOff = previewRef(id).onSnapshot(s => { if (!dead && insp) insp.setPreview(id, s && s.exists ? (s.data() || {}).html : null); }, () => {}); } catch (e) { status('Could not read the preview: ' + why(e), 'warn'); }
      };
      /* What to build a preview from: the candidate and what it needs, in dependency order, as the files the subject serves under subject/. */
      const depsOf = id => { const by = {}; merged().forEach(c => { by[c.id] = c; }); const seen = {}, out = []; const go = x => { if (seen[x] || !by[x]) return; seen[x] = true; (by[x].needs || []).forEach(go); out.push(by[x]); }; go(id); return out; };
      const buildPreview = id => {
        const e = ctx.engine(); if (!e) return Promise.reject(new Error('Sending needs the assist kit.'));
        const order = depsOf(id), c = order[order.length - 1], files = order.map(x => (x.files || []).map(f => 'subject/' + f.path).join(' ')).join(' ');
        return e.send({lane: 'task', scope: {kind: 'part', id: 'adopt.inspect', label: 'Preview of ' + id, hash: ''}, quote: c ? c.name + ' · provides ' + (c.provides || []).join(', ') : '',
          text: 'Build a preview page for candidate ' + id + ' (docs/adopt.md). Write ONE complete, self-contained HTML document that renders just this component with sample data, and nothing else of the app. Load only these files, in this order, by relative address (they are served under subject/): ' + files + ', plus the stylesheets and scripts that subject/index.html links before them (read it for the order, the theme and the sample data). Call the component\'s public API (' + ((c && c.provides) || []).join(', ') + ') the way the subject does, in a container with room to draw. Save it as the field html of the db doc ns/shared/previews/' + id + ' (collection ns/shared/previews, doc id ' + id + '). Do not change the subject or any candidate\'s state.'});
      };
      connected.then(() => fetch('subject/candidates.json')).then(r => { if (!r.ok) throw new Error(r.status); return r.json(); }).then(j => {
        if (dead) return; base = j.candidates || [];
        note.textContent = (j.subject && j.subject.name ? j.subject.name + ': ' : '') + base.length + ' candidates' + (j.namespace ? ' (namespace ' + j.namespace + ')' : '') + '.';
        /* can() answers with a promise: an editor can write the db; anyone else sees the decisions and cannot change them. */
        Promise.resolve(window.claude && claude.use ? claude.use('user') : null).then(u => (u && u.can ? u.can('data.write') : null)).catch(() => null).then(w => {
          if (dead) return; can = w !== false;
          const l = merged();
          board = Kit.adoptBoard.mount(boardEl, {candidates: l, selected: sel, packs, readOnly: !can,
            onSelect: id => choose(id, true),
            onMove: (id, to) => { const c = merged().find(x => x.id === id); if (c && Kit.adopt.can(c.state, to)) { save(id, {state: to}); if (to === 'ready') loadPack(id); } },
            onNote: (id, text) => save(id, {notes: text}),
            onWork: (id, text) => { const e = ctx.engine(); if (!e) return Promise.reject(new Error('Sending needs the assist kit.')); const c = merged().find(x => x.id === id);
              const quote = c ? c.name + ' · ' + (c.files || []).map(f => f.path).join(', ') + ' · provides ' + (c.provides || []).join(', ') : '';
              return e.send({lane: 'task', scope: {kind: 'part', id: 'adopt.board', label: 'Candidate ' + id, hash: ''}, quote,
                text: 'Adopt candidate ' + id + ' of this subject (docs/adopt.md; the candidates contract is there too). ' + text + ' When you change files, put the result in the db doc ns/shared/packs/' + id + ' (collection ns/shared/packs, an adp-pack/1 object from adp.py pack) and set state "preview" on the item with id ' + id + ' in the list ns/shared/adopt/state.items (write the whole list back).'}).then(() => { if (c && c.state !== 'working') save(id, {state: 'working'}); }); }});
          insp = Kit.adoptInspector.mount(inspEl, {src: SRC, candidates: l, selected: sel, onBuild: buildPreview, onPreviewState: reportPreview,
            /* A block nobody owns, given to a candidate: its classes and ids join that candidate's, on the server, and the candidate is selected. */
            onAssign: a => { const had = overlay[a.id] || {}, add = (x, y) => (x || []).concat((y || []).filter(k => (x || []).indexOf(k) < 0)); return save(a.id, {classes: add(had.classes, a.classes), ids: add(had.ids, a.ids)}).then(() => { choose(a.id, true); note.textContent = 'Assigned ' + a.classes.concat(a.ids.map(i => '#' + i)).slice(0, 4).join(' ') + ' to ' + a.id + '. Clicking it now selects that candidate.'; }); },
            onPick: p => { if (!p.id) { note.textContent = 'Nobody owns ' + (p.label || '<' + p.tag + '>') + '. Assign it to a candidate under the preview.'; return; } choose(p.id, true); note.textContent = p.id + ' (matched by ' + p.via + ' ' + p.value + ').'; }});
          if (!db) status('Not connected to the server: decisions will not be kept after a reload.', 'warn');
          else if (!can) status('You can look but not change: only editors can save decisions.', 'warn');
          if (ref) offs.push(ref.onSnapshot(s => {
            if (dead) return; overlay = s && s.exists ? fromItems(s.data()) : {}; push();
            const n = Object.keys(overlay).length;
            if (!restored) { restored = true; const view = can ? '' : ' You can look but not change: only editors can save decisions.';
              if (can || n) status((n ? 'Restored ' + n + ' decision' + (n === 1 ? '' : 's') + ' from the server.' : 'Nothing saved on the server yet.') + view, can ? (n ? 'ok' : '') : 'warn'); }
            Object.keys(overlay).forEach(id => { if (overlay[id].state === 'ready' || overlay[id].state === 'preview') loadPack(id); });
          }, e => status('Could not read the saved decisions: ' + why(e), 'danger')));
          if (viewRef) offs.push(viewRef.onSnapshot(s => { if (dead || sel || !s || !s.exists) return; const id = (s.data() || {}).selected; if (id && base.some(c => c.id === id)) choose(id, false); }, () => {}));      // the candidate shown last, restored once
        });
      }).catch(() => { boardEl.innerHTML = '<p class="muted">No <code>subject/candidates.json</code> beside this page. Run <code>adp.py extract SUBJECT --into DIR</code> and publish again.</p>'; inspEl.hidden = true; });
      return {destroy(){ dead = true; if (pvOff) { try { pvOff(); } catch (e) {} } offs.forEach(f => { try { f(); } catch (e) {} }); if (board) board.destroy(); if (insp) insp.destroy(); }};
    },
  },
  /* /part:adopt */
});
})();
/* /part:page.script.adopt */
