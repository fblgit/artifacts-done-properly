/* part:page.script.adopt */
/* The Adopt view (docs/adopt.md). It registers itself with the assistant's app.js through window.ADP_VIEWS.
   Files beside the page, written by `adp.py extract SUBJECT --into DIR`: subject/candidates.json (what was detected) and the subject's own files under subject/.
   The person's decisions live in one db doc, ns/shared/adopt/state: {candidate id: {state, notes, by, at}}; a ready candidate's pack is ns/shared/adopt/packs/<id>. */
(function(){
'use strict';
const SRC = 'subject/index.html', STATE = 'adopt/state';
const esc = t => String(t == null ? '' : t).replace(/[&<>"]/g, c => ({'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;'}[c]));

window.ADP_VIEWS = Object.assign(window.ADP_VIEWS || {}, {
  /* part:adopt */
  adopt: {
    label: 'Adopt',
    parts: ['adopt', 'adopt.intro', 'adopt.board', 'adopt.inspect'],
    titles: {adopt: 'Adopt', 'adopt.intro': 'Adopt intro', 'adopt.board': 'Candidates', 'adopt.inspect': 'Live preview'},
    html: () => `<section class="view adopt-view" data-part="adopt" data-comment-target aria-label="Adopt">
      <!-- part:adopt.intro --><div class="card" data-part="adopt.intro" data-comment-target><h2>Adopt</h2><p class="muted">Detect, select, work with Claude, preview, then mark each candidate ready or discard it. Pick an element in the preview or a node in the map; a ready candidate downloads as a <code>.adp.json</code> pack.</p><p class="muted ad-note" role="status"></p></div><!-- /part:adopt.intro -->
      <div class="ad-split">
        <!-- part:adopt.inspect --><div class="card ad-inspect" data-part="adopt.inspect" data-comment-target aria-label="Live preview of the artifact"></div><!-- /part:adopt.inspect -->
        <!-- part:adopt.board --><div class="card ad-board" data-part="adopt.board" data-comment-target aria-label="Candidates"></div><!-- /part:adopt.board -->
      </div>
    </section>`,
    mount(ctx){
      const Kit = window.Kit || {}, $ = s => document.querySelector(s), note = $('.ad-note'), boardEl = $('.ad-board'), inspEl = $('.ad-inspect');
      if (!Kit.adopt || !Kit.adoptBoard || !Kit.adoptInspector) { boardEl.innerHTML = '<p class="muted">The adopt kits are not installed.</p>'; return {destroy(){}}; }
      let dead = false, off = null, board = null, insp = null, base = [], overlay = {}, packs = {}, sel = null, can = true;
      const ref = window.ADP && ADP.db ? ADP.ns('shared').doc(STATE) : null, packRef = id => ADP.ns('shared').doc('adopt/packs/' + id);
      const merged = () => Kit.adopt.merge(base, overlay);
      const push = () => { const l = merged(); if (board) board.setCandidates(l); if (insp) insp.setCandidates(l); };
      const save = (id, patch) => {
        overlay = Object.assign({}, overlay, {[id]: Object.assign({}, overlay[id], patch, {at: new Date().toISOString()})}); push();
        if (!ref) { note.textContent = 'There is no database here, so decisions last until this page closes.'; return Promise.resolve(); }
        return ref.set(overlay).catch(() => { note.textContent = 'The change could not be saved.'; });
      };
      const loadPack = id => { if (!ref || packs[id] !== undefined) return; packs[id] = null; packRef(id).get().then(s => { if (!dead && s && s.exists) { packs[id] = s.data(); if (board) board.setPacks(packs); } }).catch(() => {}); };
      fetch('subject/candidates.json').then(r => { if (!r.ok) throw new Error(r.status); return r.json(); }).then(j => {
        if (dead) return; base = j.candidates || [];
        note.textContent = (j.subject && j.subject.name ? j.subject.name + ': ' : '') + base.length + ' candidates detected' + (j.namespace ? ' (namespace ' + j.namespace + ')' : '') + '.';
        Promise.resolve(window.claude && claude.use ? claude.use('user') : null).then(u => { try { can = !(u && u.can) || u.can('data.write') !== false; } catch (e) {} }).catch(() => {}).then(() => {
          if (dead) return;
          const l = merged();
          board = Kit.adoptBoard.mount(boardEl, {candidates: l, selected: sel, packs, readOnly: !can,
            onSelect: id => { sel = id; if (insp) insp.select(id); loadPack(id); },
            onMove: (id, to) => { const c = merged().find(x => x.id === id); if (c && Kit.adopt.can(c.state, to)) { save(id, {state: to}); if (to === 'ready') loadPack(id); } },
            onNote: (id, text) => save(id, {notes: text}),
            onWork: (id, text) => { const e = ctx.engine(); if (!e) return Promise.reject(new Error('Sending needs the assist kit.')); const c = merged().find(x => x.id === id);
              const quote = c ? c.name + ' · ' + (c.files || []).map(f => f.path).join(', ') + ' · provides ' + (c.provides || []).join(', ') : '';
              return e.send({lane: 'task', scope: {kind: 'part', id: 'adopt.board', label: 'Candidate ' + id, hash: ''}, quote,
                text: 'Adopt candidate ' + id + ' of this subject (docs/adopt.md). ' + text + ' When you change files, put the result in the db doc ns/shared/adopt/packs/' + id + ' (an adp-pack/1 object from adp.py pack) and move ns/shared/adopt/state ' + id + ' to "preview".'}).then(() => { if (c && c.state !== 'working') save(id, {state: 'working'}); }); }});
          insp = Kit.adoptInspector.mount(inspEl, {src: SRC, candidates: l, selected: sel,
            onPick: p => { if (!p.id) { note.textContent = 'No candidate owns <' + p.tag + '>.'; return; } sel = p.id; board.select(p.id); insp.select(p.id); loadPack(p.id); note.textContent = p.id + ' (matched by ' + p.via + ' ' + p.value + ').'; }});
          if (ref) off = ref.onSnapshot(s => { if (dead) return; overlay = s && s.exists ? s.data() : {}; push(); Object.keys(overlay).forEach(id => { if (overlay[id].state === 'ready' || overlay[id].state === 'preview') loadPack(id); }); });
        });
      }).catch(() => { boardEl.innerHTML = '<p class="muted">No <code>subject/candidates.json</code> beside this page. Run <code>adp.py extract SUBJECT --into DIR</code> and publish again.</p>'; inspEl.hidden = true; });
      return {destroy(){ dead = true; if (off) off(); if (board) board.destroy(); if (insp) insp.destroy(); }};
    },
  },
  /* /part:adopt */
});
})();
/* /part:page.script.adopt */
